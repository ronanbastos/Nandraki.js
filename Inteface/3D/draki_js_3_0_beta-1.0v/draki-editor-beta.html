Ligar esta interface ao seu draki3d.js

O editor guarda cena, UI, scripts e o loop. O desenho é delegado. Enquanto o ThreeCore real não estiver carregado, roda o rasterizador CPU embutido. Registre o core e ele assume no frame seguinte, sem tocar em nada da UI.

<!-- index.html -->
<script type="module" src="main.js"></script>

// main.js
import { DrakiEditor } from './draki3d.js';
document.addEventListener('DOMContentLoaded', () => new DrakiEditor());

// draki3d.js — o core assume o viewport deste editor
const core = new ThreeCore();
core.init(document.getElementById('viewport'));

Draki.setCore({
  init(canvas, api) {
    this.r = new THREE.WebGLRenderer({ canvas, antialias: true });
    this.scene = new THREE.Scene();
    this.cam = new THREE.PerspectiveCamera(50, 1, .1, 2000);
    this.map = new Map();            // id do editor -> THREE.Object3D
  },
  sync(objects) { /* Game.create() para cada objeto novo */ },
  update(obj)   { /* position / rotation / scale / material */ },
  remove(id)    { this.map.get(id)?.removeFromParent(); },
  resize(w, h)  { this.r.setSize(w, h, false); this.cam.aspect = w / h; },
  frame(cam, dt){ this.r.render(this.scene, this.cam); },
  pick(x, y)    { /* Raycaster -> id do editor ou null */ }
});
Game.create(type)cube, sphere, plane, cylinder, cone, torus, ring, camera, light.
Draki.scene()Array plano com parent, transform, material e components.
Draki.selectedObjectObjeto ativo no inspetor, ou null.
ScriptComponent(src)Script por objeto, roda a cada frame no Play.
animax(obj, props, dur)Tween com ease-out, disponível dentro dos scripts.
input.joystick(name)Vetor {x,y} normalizado do joystick da HUD.
input.button(name)Estado pressionado dos touch / click buttons.
Draki.saveScene()Serializa tudo em .draki.json, com scripts inclusos.
